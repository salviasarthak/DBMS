 --- EXPLICIT CURSOR 
 DECLARE 
 CURSOR EMP_CURSOR IS SELECT EMPLOYEE_ID, SALARY FROM EMP;

 C_EMPLOYEE_ID EMP.EMPLOYEE_ID%TYPE;
 C_SALARY EMP.SALARY%TYPE;
 SALARY_ZERO EXCEPTION;
 BEGIN
    OPEN EMP_CURSOR;
    FETCH EMP_CURSOR INTO C_EMPLOYEE_ID , C_SALARY;
    WHILE EMP_CURSOR%FOUND
    LOOP
        IF C_SALARY = 0 THEN 
        RAISE SALARY_ZERO;
        END IF;
    UPDATE EMP SET SALARY = C_SALARY * 1.10
    WHERE EMPLOYEE_ID = C_EMPLOYEE_ID;

    FETCH EMP_CURSOR INTO C_EMPLOYEE_ID , C_SALARY;
    END LOOP;
    CLOSE EMP_CURSOR;

    EXCEPTION WHEN SALARY_ZERO THEN 
    DBMS_OUTPUT.PUT_LINE('ZERO SAL!');
END;

 -- SHOWS UPDATED VALUE
 SELECT * FROM EMP;
